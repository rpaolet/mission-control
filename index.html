Configurazione Automatica Google Sheets (Apps Script)

  1. Crea un nuovo Foglio Google.
  2. Nel menu in alto, clicca su Estensioni > Apps Script.
  3. Elimina tutto il codice esistente e incolla il codice sottostante:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Se è la prima riga, aggiungi l'intestazione
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Data Inserimento", "Nome", "Email", "Categoria", "Data", "Note"]);
    }
    
    sheet.appendRow([
      new Date().toLocaleString(),
      data.name,
      data.email,
      data.category,
      data.date,
      data.notes || ""
    ]);
    
    return ContentService
      .createTextOutput(JSON.stringify({ "status": "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
